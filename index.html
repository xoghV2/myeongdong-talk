const ADMIN_EMAIL = 'rktwlshbbb@gmail.com';
const ADMIN_PASS = '1234'; // 원하는 비밀번호로 변경

function doPost(e) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(['id','date','category','message','imageBase64','status']);
  }
  const body = JSON.parse(e.postData.contents);
  if (body.action === 'update_status') {
    if (body.password !== ADMIN_PASS) return jsonOut({ ok: false, error: 'auth' });
    const rows = sheet.getDataRange().getValues();
    for (let i = 1; i < rows.length; i++) {
      if (String(rows[i][0]) === String(body.id)) {
        sheet.getRange(i + 1, 6).setValue(body.status);
        break;
      }
    }
    return jsonOut({ ok: true });
  }
  const id = String(Date.now());
  const dateStr = Utilities.formatDate(new Date(), 'Asia/Seoul', 'yyyy-MM-dd HH:mm');
  sheet.appendRow([id, dateStr, body.category, body.message, body.imageBase64 || '', '미확인']);

  // 3차 보호: 메일에는 원본 사진 없이 텍스트 요약 + 안내만 발송
  const hasImg = body.imageBase64 ? '있음 (관리자 게시판에서 모자이크 해제 후 확인)' : '없음';
  MailApp.sendEmail({
    to: ADMIN_EMAIL,
    subject: '[명동톡 문의] ' + body.category,
    body: '새 문의가 접수되었습니다.\n\n분류: ' + body.category + '\n시간: ' + dateStr + '\n첨부사진: ' + hasImg + '\n\n내용:\n' + body.message + '\n\n※ 홈페이지 [문의·설정 → 관리자 비밀 게시판]에서 확인하세요.'
  });
  return jsonOut({ ok: true, id: id, date: dateStr });
}

function doGet(e) {
  if ((e.parameter.password || '') !== ADMIN_PASS) return jsonOut({ ok: false, error: 'auth' });
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  const rows = sheet.getDataRange().getValues();
  const items = rows.slice(1).map(r => ({
    id: String(r[0]), date: String(r[1]), category: String(r[2]),
    message: String(r[3]), imageBase64: String(r[4] || ''), status: String(r[5] || '미확인')
  })).reverse();
  return jsonOut({ ok: true, items: items });
}

function jsonOut(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}
